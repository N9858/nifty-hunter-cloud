import requests, os
from flask import Flask
import time

app = Flask(__name__)

TOKEN = os.environ.get("TOKEN", "")
CHAT_ID = os.environ.get("CHAT_ID", "")

def send_telegram(msg):
    if not TOKEN or not CHAT_ID:
        print("Token/Chat ID ledu")
        return
    url = f"https://api.telegram.org/bot{TOKEN}/sendMessage"
    try:
        requests.post(url, data={"chat_id": CHAT_ID, "text": msg, "parse_mode": "HTML"})
        print("Telegram sent!")
    except Exception as e:
        print(e)

def get_nifty_data():
    try:
        # Previous day High Low from Yahoo
        url = "https://query1.finance.yahoo.com/v8/finance/chart/%5ENSEI?interval=1d&range=2d"
        r = requests.get(url, headers={'User-Agent': 'Mozilla/5.0'}, timeout=10).json()
        result = r['chart']['result'][0]
        high = result['indicators']['quote'][0]['high'][-2] # yesterday high
        low = result['indicators']['quote'][0]['low'][-2] # yesterday low
        close = result['indicators']['quote'][0]['close'][-2]

        # Current price
        curr = float(result['meta']['regularMarketPrice'])
        return high, low, close, curr
    except Exception as e:
        print("Data error", e)
        return None, None, None, None

@app.route('/')
def home():
    high, low, close, curr = get_nifty_data()
    if not high:
        return "Data loading error, refresh again"

    # Logic - Simple & Powerful
    buy_level = round(high + 10) # Yesterday High break = Buy
    sell_level = round(low - 10) # Yesterday Low break = Sell
    sl_buy = round(buy_level - 40)
    sl_sell = round(sell_level + 40)

    msg = f"""<b>📊 NIFTY DAILY ENTRY - {time.strftime('%d-%m-%Y')}</b>

💰 <b>Current: {curr}</b>
Y-High: {high} | Y-Low: {low}

🟢 <b>BUY ENTRY: {buy_level}</b>
   SL: {sl_buy}
   Target: {buy_level + 80}

🔴 <b>SELL ENTRY: {sell_level}</b>
   SL: {sl_sell}
   Target: {sell_level - 80}

<i>Buy = {buy_level} paina sustain ayithe
Sell = {sell_level} kinda sustain ayithe</i>
"""
    send_telegram(msg)
    return msg

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=10000)
